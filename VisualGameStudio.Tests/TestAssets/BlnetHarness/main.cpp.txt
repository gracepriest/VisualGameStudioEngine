/* main.cpp - blnet ABI conformance harness (Task 9+).
 *
 * Native C++ program that LoadLibrary's the Native-AOT published BlnetTestShim.dll,
 * binds the 7 contract exports (by the BLNET_EXPORT_* macro names so a header/shim
 * name drift breaks loudly here) plus the shim's test exports, initializes the
 * boundary (vtable handshake), then runs ONE scenario named by argv[1]. Exception:
 * the version_mismatch scenario dispatches BEFORE the shared handshake and performs
 * its own (mismatched-then-correct) initialize sequence.
 *
 * Exit protocol: 0 = PASS (stdout starts with "PASS <scenario>"),
 *                1 = scenario failure ("FAIL <scenario>: <detail>"),
 *                2 = environmental failure (load/bind/init handshake).
 * Each scenario runs in a FRESH process (the fixture spawns one per TestCase), so
 * shim static state never leaks between scenarios.
 *
 * Checked in as .txt so nothing compiles it in place; the conformance fixture copies
 * it to a temp dir as main.cpp next to the emitted blnet.h / blnet_runtime.hpp.
 * ASCII-only: non-ASCII test data is written as \xNN UTF-8 byte escapes.
 */
#include "blnet.h"
#include "blnet_runtime.hpp"
#include <windows.h>
#include <atomic>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <stdexcept>
#include <string>
#include <thread>

using namespace BasicLang::blnet;

/* Test-export signatures (beyond the contract, shim test surface) */
static int32_t (BLNET_CALL *test_create_list)(uint64_t*);
static int32_t (BLNET_CALL *test_list_add)(uint64_t, int32_t);
static int32_t (BLNET_CALL *test_list_count)(uint64_t, int32_t*);
static int32_t (BLNET_CALL *test_echo)(const char*, char**);
static int32_t (BLNET_CALL *test_throw)(void);
static int32_t (BLNET_CALL *test_invoke)(uint64_t, uint64_t*, int32_t, uint64_t*);
static int32_t (BLNET_CALL *test_invoke_from_thread)(uint64_t, uint64_t*, int32_t);
static int32_t (BLNET_CALL *shim_initialize)(int32_t, const BlnetNativeVtable*);
static int32_t (BLNET_CALL *shim_abi_version)(void);

/* use only in scenario functions / main - inside NativeCallbackFn lambdas, set a flag and
   REQUIRE it after the invoke (return-from-macro would only exit the callback) */
#define REQUIRE(cond, name, detail) \
    do { if (!(cond)) { printf("FAIL %s: %s\n", name, detail); return 1; } } while (0)

template <typename T> static void bind(HMODULE m, T& fn, const char* name) {
    fn = reinterpret_cast<T>(GetProcAddress(m, name));
    if (!fn) { printf("FAIL bind: missing export %s\n", name); exit(2); }
}

/* ---- scenario 1: create -> add -> count round-trips a live handle ---- */
static int scenario_handle_roundtrip() {
    uint64_t h = 0;
    { BlnetCallScope s; NetCheck(test_create_list(&h)); }
    NetRef ref(h);
    { BlnetCallScope s; NetCheck(test_list_add(ref.get(), 42)); }
    int32_t count = 0;
    { BlnetCallScope s; NetCheck(test_list_count(ref.get(), &count)); }
    REQUIRE(count == 1, "handle_roundtrip", "count != 1");
    printf("PASS handle_roundtrip\n");
    return 0;
}

/* ---- scenario 2: use-after-release RETURNS BLNET_E_STALE_HANDLE (never crashes) ---- */
static int scenario_stale_handle() {
    uint64_t h = 0;
    { BlnetCallScope s; NetCheck(test_create_list(&h)); }
    { NetRef ref(h); } /* scope exit -> blnet_release -> refcount 0 -> slot freed */
    int32_t st;
    { BlnetCallScope s; st = test_list_add(h, 1); } /* raw status: failure is EXPECTED here */
    REQUIRE(st == BLNET_E_STALE_HANDLE, "stale_handle", "dead-handle use did not return BLNET_E_STALE_HANDLE");
    printf("PASS stale_handle\n");
    return 0;
}

/* ---- scenario 3: refcount protocol - addref/release pair, aliveness tracks the
   count, and the release AFTER free is stale (double-release detection) ---- */
static int scenario_double_release_addref() {
    uint64_t h = 0;
    { BlnetCallScope s; NetCheck(test_create_list(&h)); }        /* born refcount 1 */
    { BlnetCallScope s; NetCheck(g_shim.addref(h)); }            /* refcount 2 */
    int32_t st, count = 0;
    { BlnetCallScope s; st = g_shim.release(h); }                /* refcount 1 */
    REQUIRE(st == BLNET_OK, "double_release_addref", "first release failed");
    { BlnetCallScope s; st = test_list_count(h, &count); }
    REQUIRE(st == BLNET_OK, "double_release_addref", "handle dead after first release (refcount still 1)");
    { BlnetCallScope s; st = g_shim.release(h); }                /* refcount 0 -> freed */
    REQUIRE(st == BLNET_OK, "double_release_addref", "second release failed");
    { BlnetCallScope s; st = test_list_count(h, &count); }
    REQUIRE(st == BLNET_E_STALE_HANDLE, "double_release_addref", "handle still alive after refcount hit zero");
    { BlnetCallScope s; st = g_shim.release(h); }                /* third release: past zero */
    REQUIRE(st == BLNET_E_STALE_HANDLE, "double_release_addref", "release past zero did not return BLNET_E_STALE_HANDLE");
    printf("PASS double_release_addref\n");
    return 0;
}

/* ---- scenario 4: slot reuse bumps the generation - the recycled slot's new handle
   works while the old handle stays stale (C2 generation tagging).
   In this fresh process the first create takes index 1 (index 0 is burned); after
   its release the next create reuses index 1 -> same low-32, different high-32. ---- */
static int scenario_generation_reuse() {
    uint64_t h1 = 0;
    { BlnetCallScope s; NetCheck(test_create_list(&h1)); }
    { BlnetCallScope s; NetCheck(g_shim.release(h1)); }          /* freed -> slot to freelist */
    uint64_t h2 = 0;
    { BlnetCallScope s; NetCheck(test_create_list(&h2)); }       /* reuses the slot */
    REQUIRE((h1 & 0xFFFFFFFFull) == (h2 & 0xFFFFFFFFull), "generation_reuse", "slot index was not reused");
    REQUIRE(h1 != h2, "generation_reuse", "generation did not change on slot reuse");
    int32_t st, count = -1;
    { BlnetCallScope s; st = test_list_add(h1, 7); }             /* old handle must stay dead */
    REQUIRE(st == BLNET_E_STALE_HANDLE, "generation_reuse", "old handle usable after slot reuse");
    { BlnetCallScope s; NetCheck(test_list_add(h2, 7)); }        /* new handle must work */
    { BlnetCallScope s; NetCheck(test_list_count(h2, &count)); }
    REQUIRE(count == 1, "generation_reuse", "new handle count != 1");
    { BlnetCallScope s; NetCheck(g_shim.release(h2)); }
    printf("PASS generation_reuse\n");
    return 0;
}

/* ---- scenario 5: UTF-8 string crosses the boundary byte-for-byte both ways (C3).
   Input is "hello world" with e-acute, o-umlaut and a check mark, written as \xNN
   UTF-8 byte escapes (narrow literal, NEVER u8"..."; keeps the source ASCII-clean
   and removes any compiler source-encoding dependency). Output buffer comes from
   blnet_alloc inside the shim; the receiver frees it via blnet_free. ---- */
static int scenario_string_roundtrip() {
    const char* input = "h\xC3\xA9llo w\xC3\xB6rld \xE2\x9C\x93";
    char* out = nullptr;
    int32_t st;
    { BlnetCallScope s; st = test_echo(input, &out); }
    REQUIRE(st == BLNET_OK, "string_roundtrip", "echo failed");
    REQUIRE(out != nullptr, "string_roundtrip", "echo returned a null buffer");
    const bool same = std::strcmp(out, input) == 0;
    g_shim.free_(out);
    REQUIRE(same, "string_roundtrip", "echoed bytes differ from the input");
    printf("PASS string_roundtrip\n");
    return 0;
}

static bool contains(const char* haystack, const char* needle) {
    return haystack && std::strstr(haystack, needle) != nullptr;
}

/* ---- scenario 6: a managed exception surfaces as BLNET_E_MANAGED_EXCEPTION with
   details via blnet_last_error (C4). The shim throws ArgumentException("b\xC3\xB8\xC3\xB8m
   from managed") - "boom" with o-slash UTF-8 bytes. Two lenses, in this order:
   (a) raw status + our own blnet_last_error inspection FIRST (NetCheck itself calls
   blnet_last_error, so keep the raw-path inspection independent of it), then
   (b) NetCheck on a FRESH throw -> std::runtime_error carrying type + message. ---- */
static int scenario_managed_exception() {
    int32_t st;
    { BlnetCallScope s; st = test_throw(); }
    REQUIRE(st == BLNET_E_MANAGED_EXCEPTION, "managed_exception", "test_throw did not return BLNET_E_MANAGED_EXCEPTION");
    char* type = nullptr; char* msg = nullptr;
    REQUIRE(g_shim.last_error(&type, &msg) == BLNET_OK, "managed_exception", "blnet_last_error failed");
    const bool type_ok = contains(type, "ArgumentException");
    const bool msg_ok  = contains(msg, "b\xC3\xB8\xC3\xB8m");
    if (type) g_shim.free_(type);
    if (msg)  g_shim.free_(msg);
    REQUIRE(type_ok, "managed_exception", "last-error type does not contain ArgumentException");
    REQUIRE(msg_ok,  "managed_exception", "last-error message does not contain the boom UTF-8 bytes");

    bool threw = false;
    std::string what;
    try {
        BlnetCallScope s;
        NetCheck(test_throw());
    } catch (const std::runtime_error& ex) { threw = true; what = ex.what(); }
    REQUIRE(threw, "managed_exception", "NetCheck did not throw on BLNET_E_MANAGED_EXCEPTION");
    REQUIRE(what.find("ArgumentException") != std::string::npos, "managed_exception", "NetCheck what() lacks the exception type");
    REQUIRE(what.find("b\xC3\xB8\xC3\xB8m") != std::string::npos, "managed_exception", "NetCheck what() lacks the boom UTF-8 bytes");
    printf("PASS managed_exception\n");
    return 0;
}

/* ---- scenario 7: a native exception inside an inline callback surfaces as
   BLNET_E_NATIVE_EXCEPTION (C4/C5). The callback body throws "n\xC3\xA4tive boom"
   ("native" with a-umlaut UTF-8 bytes); the invoke runs inline because the harness
   wraps test_invoke in BlnetCallScope (depth > 0 -> same-thread dispatch). The shim
   pulls the message across via get_native_error and re-exposes it through
   blnet_last_error under the synthetic type BasicLangNativeException. ---- */
static int scenario_inline_native_exception() {
    blnet_callback cb = blnet_register_callback(
        [](const uint64_t*, int32_t, uint64_t*) -> int32_t {
            throw std::runtime_error("n\xC3\xA4tive boom");
        },
        nullptr, 0, CallbackFlags{});
    int32_t st;
    { BlnetCallScope s; st = test_invoke(cb, nullptr, 0, nullptr); }
    REQUIRE(st == BLNET_E_NATIVE_EXCEPTION, "inline_native_exception", "inline throw did not return BLNET_E_NATIVE_EXCEPTION");
    char* type = nullptr; char* msg = nullptr;
    REQUIRE(g_shim.last_error(&type, &msg) == BLNET_OK, "inline_native_exception", "blnet_last_error failed");
    const bool type_ok = contains(type, "BasicLangNativeException");
    const bool msg_ok  = contains(msg, "n\xC3\xA4tive boom");
    if (type) g_shim.free_(type);
    if (msg)  g_shim.free_(msg);
    REQUIRE(type_ok, "inline_native_exception", "last-error type does not contain BasicLangNativeException");
    REQUIRE(msg_ok,  "inline_native_exception", "last-error message lacks the native boom UTF-8 bytes");
    printf("PASS inline_native_exception\n");
    return 0;
}

/* ---- scenario 8: inline result-bearing callback with a VALUE and an OUT slot (C5).
   Inline (BlnetCallScope around test_invoke) so result-bearing is allowed. The body
   reads args[0] as an int, doubles it into *result, and writes arg0+1 through the
   caller's OUT pointer in args[1]. ---- */
static int scenario_inline_result_and_out() {
    BlnetSlotDesc slots[] = { {BLNET_SLOT_VALUE, 0}, {BLNET_SLOT_OUT, 4} };
    CallbackFlags flags; flags.result_bearing = true;
    blnet_callback cb = blnet_register_callback(
        [](const uint64_t* args, int32_t, uint64_t* result) -> int32_t {
            const int32_t v = static_cast<int32_t>(args[0]);
            *result = static_cast<uint64_t>(v * 2);
            *reinterpret_cast<int32_t*>(args[1]) = v + 1;
            return BLNET_OK;
        },
        slots, 2, flags);
    int32_t out_val = 0;
    uint64_t args[2] = { 21, reinterpret_cast<uint64_t>(&out_val) };
    uint64_t result = 0;
    int32_t st;
    { BlnetCallScope s; st = test_invoke(cb, args, 2, &result); }
    REQUIRE(st == BLNET_OK, "inline_result_and_out", "inline result-bearing invoke failed");
    REQUIRE(result == 42, "inline_result_and_out", "result slot != 42");
    REQUIRE(out_val == 22, "inline_result_and_out", "out-slot int != 22");
    printf("PASS inline_result_and_out\n");
    return 0;
}

/* ---- scenario 9: a 16-byte struct crosses via BLNET_SLOT_STRUCT (borrowed pointer)
   and comes back through a caller-allocated out-buffer whose ADDRESS is pre-loaded
   into the result slot. Values are exactly representable doubles so the round trip
   is bit-exact under operator==. ---- */
static int scenario_struct_slots() {
    struct Pt { double x; double y; };
    static_assert(sizeof(Pt) == 16, "Pt must be exactly 16 bytes");
    BlnetSlotDesc slots[] = { {BLNET_SLOT_STRUCT, 16} };
    CallbackFlags flags; flags.result_bearing = true;
    blnet_callback cb = blnet_register_callback(
        [](const uint64_t* args, int32_t, uint64_t* result) -> int32_t {
            const Pt* in = reinterpret_cast<const Pt*>(args[0]);
            Pt* out = reinterpret_cast<Pt*>(*result); /* caller pre-loaded the out-buffer address */
            out->x = in->x * 2; out->y = in->y * 2;
            return BLNET_OK;
        },
        slots, 1, flags);
    Pt in_pt{1.5, 2.25};
    Pt out_pt{0.0, 0.0};
    uint64_t args[1] = { reinterpret_cast<uint64_t>(&in_pt) };
    uint64_t result = reinterpret_cast<uint64_t>(&out_pt);
    int32_t st;
    { BlnetCallScope s; st = test_invoke(cb, args, 1, &result); }
    REQUIRE(st == BLNET_OK, "struct_slots", "inline struct invoke failed");
    REQUIRE(out_pt.x == 3.0, "struct_slots", "out struct x != 3.0 (bit-exact)");
    REQUIRE(out_pt.y == 4.5, "struct_slots", "out struct y != 4.5 (bit-exact)");
    printf("PASS struct_slots\n");
    return 0;
}

/* ---- scenario 10: a cross-thread notification with a HANDLE slot is queued, and the
   queue's addref keeps the object alive across the caller's release until the pump
   runs it (C5/C2). The shim invokes the thunk on a fresh .NET thread (depth 0 there
   -> queued) and JOINS it, so by the time test_invoke_from_thread returns the
   invocation is enqueued but NOT executed. Globals (s10_*) carry the callback's
   observations out; the callback body never REQUIREs. All s10_ writes happen on the
   pump thread (== main), so plain statics are race-free. ---- */
static bool    s10_ran = false;
static int32_t s10_status = -1;
static int32_t s10_count = -1;

static int scenario_queued_notification_addref() {
    BlnetSlotDesc slots[] = { {BLNET_SLOT_HANDLE, 0} };
    blnet_callback cb = blnet_register_callback(
        [](const uint64_t* args, int32_t, uint64_t*) -> int32_t {
            int32_t count = 0; int32_t st;
            { BlnetCallScope s; st = test_list_count(args[0], &count); }
            s10_status = st; s10_count = count; s10_ran = true;
            return BLNET_OK;
        },
        slots, 1, CallbackFlags{});
    uint64_t h = 0;
    { BlnetCallScope s; NetCheck(test_create_list(&h)); }
    {
        NetRef ref(h);
        { BlnetCallScope s; NetCheck(test_list_add(ref.get(), 42)); }
        uint64_t inv_args[1] = { ref.get() };
        int32_t st;
        { BlnetCallScope s; st = test_invoke_from_thread(cb, inv_args, 1); }
        REQUIRE(st == BLNET_OK, "queued_notification_addref", "cross-thread notification enqueue failed");
        REQUIRE(!s10_ran, "queued_notification_addref", "notification ran before blnet_pump");
    } /* NetRef released HERE - the queue's addref is now the only ref keeping the list alive */
    int32_t pump_st = blnet_pump();
    REQUIRE(pump_st == BLNET_OK, "queued_notification_addref", "blnet_pump failed");
    REQUIRE(s10_ran, "queued_notification_addref", "notification did not run during blnet_pump");
    REQUIRE(s10_status == BLNET_OK, "queued_notification_addref", "handle was dead inside the pumped callback (queue addref missing)");
    REQUIRE(s10_count == 1, "queued_notification_addref", "count inside the pumped callback != 1");
    int32_t st, count = 0;
    { BlnetCallScope s; st = test_list_count(h, &count); }
    REQUIRE(st == BLNET_E_STALE_HANDLE, "queued_notification_addref", "handle still alive after the pump (queue release missing)");
    printf("PASS queued_notification_addref\n");
    return 0;
}

/* ---- scenario 11: a result-bearing callback (without Immediate) invoked cross-thread
   is REJECTED with BLNET_E_CROSS_THREAD_RESULT - never run, never queued (C5/C6).
   The shim's Join gives happens-before, so reading s11_ran after the call is safe. ---- */
static bool s11_ran = false;

static int scenario_cross_thread_result_rejected() {
    CallbackFlags flags; flags.result_bearing = true;
    blnet_callback cb = blnet_register_callback(
        [](const uint64_t*, int32_t, uint64_t*) -> int32_t { s11_ran = true; return BLNET_OK; },
        nullptr, 0, flags);
    int32_t st;
    { BlnetCallScope s; st = test_invoke_from_thread(cb, nullptr, 0); }
    REQUIRE(st == BLNET_E_CROSS_THREAD_RESULT, "cross_thread_result_rejected", "cross-thread result-bearing invoke was not rejected");
    REQUIRE(!s11_ran, "cross_thread_result_rejected", "rejected callback still ran");
    int32_t pump_st = blnet_pump();
    REQUIRE(pump_st == BLNET_OK, "cross_thread_result_rejected", "blnet_pump failed after the rejection");
    REQUIRE(!s11_ran, "cross_thread_result_rejected", "rejected callback was queued anyway (ran on pump)");
    printf("PASS cross_thread_result_rejected\n");
    return 0;
}

/* ---- scenario 12: the pump CONTINUES past failing notifications, fires the error
   hook per failure (queue order preserved), and returns the FIRST failure's status;
   a drained queue pumps clean afterwards (C4/C5). The hook must be a capture-less
   lambda (raw function pointer), so its observations go through globals. ---- */
static bool s12_first_ran = false;
static int  s12_hook_fires = 0;
static int  s12_hook_bad_status = 0; /* fires whose status != BLNET_E_NATIVE_EXCEPTION */
static std::string s12_hook_msgs;

static int scenario_pump_error_surfacing() {
    blnet_callback cb1 = blnet_register_callback(
        [](const uint64_t*, int32_t, uint64_t*) -> int32_t { s12_first_ran = true; return BLNET_OK; },
        nullptr, 0, CallbackFlags{});
    blnet_callback cb2 = blnet_register_callback(
        [](const uint64_t*, int32_t, uint64_t*) -> int32_t { throw std::runtime_error("boom2"); },
        nullptr, 0, CallbackFlags{});
    blnet_callback cb3 = blnet_register_callback(
        [](const uint64_t*, int32_t, uint64_t*) -> int32_t { throw std::runtime_error("boom3"); },
        nullptr, 0, CallbackFlags{});
    int32_t st;
    { BlnetCallScope s; st = test_invoke_from_thread(cb1, nullptr, 0); }
    REQUIRE(st == BLNET_OK, "pump_error_surfacing", "enqueue #1 failed");
    { BlnetCallScope s; st = test_invoke_from_thread(cb2, nullptr, 0); }
    REQUIRE(st == BLNET_OK, "pump_error_surfacing", "enqueue #2 failed");
    { BlnetCallScope s; st = test_invoke_from_thread(cb3, nullptr, 0); }
    REQUIRE(st == BLNET_OK, "pump_error_surfacing", "enqueue #3 failed");
    /* plain counters are fine: the hook runs on the pump thread only (== main) */
    blnet_set_error_hook([](int32_t hook_st, const char* msg) {
        ++s12_hook_fires;
        if (hook_st != BLNET_E_NATIVE_EXCEPTION) ++s12_hook_bad_status;
        s12_hook_msgs += msg ? msg : "<null>";
        s12_hook_msgs += ";";
    });
    int32_t pump_st = blnet_pump();
    REQUIRE(pump_st == BLNET_E_NATIVE_EXCEPTION, "pump_error_surfacing", "pump did not return the first failure status");
    REQUIRE(s12_first_ran, "pump_error_surfacing", "callback #1 did not run");
    REQUIRE(s12_hook_fires == 2, "pump_error_surfacing", "error hook did not fire exactly twice");
    REQUIRE(s12_hook_bad_status == 0, "pump_error_surfacing", "error hook fired with a status other than BLNET_E_NATIVE_EXCEPTION");
    const size_t p2 = s12_hook_msgs.find("boom2");
    const size_t p3 = s12_hook_msgs.find("boom3");
    REQUIRE(p2 != std::string::npos, "pump_error_surfacing", "hook messages lack boom2");
    REQUIRE(p3 != std::string::npos, "pump_error_surfacing", "hook messages lack boom3");
    REQUIRE(p2 < p3, "pump_error_surfacing", "boom2 did not precede boom3 (queue order broken)");
    int32_t pump2 = blnet_pump();
    REQUIRE(pump2 == BLNET_OK, "pump_error_surfacing", "second pump on a drained queue was not OK");
    REQUIRE(s12_hook_fires == 2, "pump_error_surfacing", "hook fired again on a drained queue");
    printf("PASS pump_error_surfacing\n");
    return 0;
}

/* ---- scenario 13: enqueue never blocks on the pump (deadlock guard, C6). We are ON
   the pump thread; the shim spawns a .NET thread whose invoke queues the notification
   and then JOINS that thread before returning. A design where enqueue blocked until
   the pump ran the callback would deadlock right here (pump thread waiting on Join,
   spawned thread waiting on a pump that can never run) and the fixture's 60 s timeout
   would kill the process - so merely RETURNING is the property under test. ---- */
static bool s13_ran = false;

static int scenario_deadlock_guard() {
    blnet_callback cb = blnet_register_callback(
        [](const uint64_t*, int32_t, uint64_t*) -> int32_t { s13_ran = true; return BLNET_OK; },
        nullptr, 0, CallbackFlags{});
    int32_t st;
    { BlnetCallScope s; st = test_invoke_from_thread(cb, nullptr, 0); }
    REQUIRE(st == BLNET_OK, "deadlock_guard", "cross-thread notification enqueue failed");
    REQUIRE(!s13_ran, "deadlock_guard", "notification ran before blnet_pump");
    int32_t pump_st = blnet_pump();
    REQUIRE(pump_st == BLNET_OK, "deadlock_guard", "blnet_pump failed");
    REQUIRE(s13_ran, "deadlock_guard", "notification did not run during blnet_pump");
    printf("PASS deadlock_guard\n");
    return 0;
}

/* ---- scenario 14: version fail-fast (C7). The MISMATCHED initialize must be the
   FIRST initialize call in this process (main special-cases the dispatch so the
   shared handshake never ran), and the shim checks the version BEFORE the null
   guard and the vtable read, so the rejected call leaves no state behind; the
   correct-version init then succeeds. BLNET_ABI_VERSION is used as a MACRO, never
   a numeric literal: a BlnetContract.AbiVersion bump mechanically breaks the drift
   test AND this scenario until both sides regenerate (C7's intent). Each scenario
   is a fresh process, so the failed init cannot poison other scenarios. ---- */
static int scenario_version_mismatch() {
    static BlnetNativeVtable vt{ &blnet_invoke_callback, &blnet_get_native_error };
    int32_t st = shim_initialize(BLNET_ABI_VERSION + 998, &vt);
    REQUIRE(st == BLNET_E_VERSION_MISMATCH, "version_mismatch", "mismatched init did not return BLNET_E_VERSION_MISMATCH");
    st = shim_initialize(BLNET_ABI_VERSION, &vt);
    REQUIRE(st == BLNET_OK, "version_mismatch", "correct-version init failed after the rejected one");
    REQUIRE(shim_abi_version() == BLNET_ABI_VERSION, "version_mismatch", "shim_abi_version() != BLNET_ABI_VERSION");
    printf("PASS version_mismatch\n");
    return 0;
}

/* ---- scenario 15: 8 threads x 2000 iterations of create->add->count->release
   through the shim, concurrently (C1/C2: the handle table under contention). Each
   thread body opens its OWN BlnetCallScope (the depth counter is thread_local - one
   scope per thread, never shared). Tallies are std::atomic: the s10-s13 plain-static
   pattern relies on pump-thread-only writes / Join happens-before, which does NOT
   hold for 8 concurrent writers. Workers never print (interleaving) and never
   REQUIRE (return-from-macro would only exit the lambda); they only count. Count
   correctness is checked per iteration (each iteration's fresh list must count
   exactly the 1 element added), not just statuses. After joining, a final
   create/add/count/release round-trip proves the table is still coherent. ---- */
static int scenario_concurrency_hammer() {
    constexpr int kThreads = 8;
    constexpr int kIters = 2000;
    std::atomic<int> status_failures{0};  /* any non-OK status from any shim call */
    std::atomic<int> count_mismatches{0}; /* count != adds performed in that iteration */
    std::atomic<int> first_bad_status{0}; /* first non-OK status observed (0 = none; BLNET_OK == 0) */
    std::thread threads[kThreads];
    for (int t = 0; t < kThreads; ++t) {
        threads[t] = std::thread([&status_failures, &count_mismatches, &first_bad_status]() {
            BlnetCallScope scope; /* thread_local depth: covers all this thread's shim calls */
            auto bad = [&](int32_t st) {
                ++status_failures;
                int expected = 0;
                first_bad_status.compare_exchange_strong(expected, static_cast<int>(st));
            };
            for (int i = 0; i < kIters; ++i) {
                uint64_t h = 0;
                int32_t st = test_create_list(&h);
                if (st != BLNET_OK) { bad(st); continue; }
                st = test_list_add(h, i);
                const bool added = st == BLNET_OK;
                if (!added) bad(st);
                int32_t count = -1;
                st = test_list_count(h, &count);
                if (st != BLNET_OK) bad(st);
                /* tally independence: only judge the count when the add succeeded,
                   so one add failure cannot bump both tallies */
                else if (added && count != 1) ++count_mismatches;
                st = g_shim.release(h);
                if (st != BLNET_OK) bad(st);
            }
        });
    }
    for (auto& th : threads) th.join();
    /* failure path only: surface the evidence the static FAIL strings discard
       (fixture pattern-matches PASS <name>; extra lines before a FAIL just show
       up in the NUnit failure message) */
    if (status_failures.load() != 0 || count_mismatches.load() != 0)
        printf("note: status_failures=%d count_mismatches=%d first_bad_status=%d\n",
            status_failures.load(), count_mismatches.load(), first_bad_status.load());
    REQUIRE(status_failures.load() == 0, "concurrency_hammer", "non-OK statuses under concurrent create/add/count/release");
    REQUIRE(count_mismatches.load() == 0, "concurrency_hammer", "per-iteration count != 1 under concurrency (cross-handle interference)");
    uint64_t h = 0;
    { BlnetCallScope s; NetCheck(test_create_list(&h)); }
    int32_t count = -1;
    { BlnetCallScope s; NetCheck(test_list_add(h, 1)); }
    { BlnetCallScope s; NetCheck(test_list_count(h, &count)); }
    REQUIRE(count == 1, "concurrency_hammer", "post-join round-trip count != 1");
    { BlnetCallScope s; NetCheck(g_shim.release(h)); }
    printf("PASS concurrency_hammer\n");
    return 0;
}

/* ---- scenario 16: releasing a callback makes its handle stale (C5, mirroring the
   C2 handle rules). blnet_callback_release bumps the slot generation, so a later
   inline invoke through the universal thunk returns BLNET_E_STALE_CALLBACK without
   running the body, and a second release is stale too (double-release detection). ---- */
static bool s16_ran = false;

static int scenario_stale_callback_via_release() {
    blnet_callback cb = blnet_register_callback(
        [](const uint64_t*, int32_t, uint64_t*) -> int32_t { s16_ran = true; return BLNET_OK; },
        nullptr, 0, CallbackFlags{});
    REQUIRE(blnet_callback_release(cb) == BLNET_OK, "stale_callback_via_release", "first release failed");
    int32_t st;
    { BlnetCallScope s; st = test_invoke(cb, nullptr, 0, nullptr); }
    REQUIRE(st == BLNET_E_STALE_CALLBACK, "stale_callback_via_release", "invoke on a released callback did not return BLNET_E_STALE_CALLBACK");
    REQUIRE(!s16_ran, "stale_callback_via_release", "released callback body still ran");
    REQUIRE(blnet_callback_release(cb) == BLNET_E_STALE_CALLBACK, "stale_callback_via_release", "double release did not return BLNET_E_STALE_CALLBACK");
    printf("PASS stale_callback_via_release\n");
    return 0;
}

int main(int argc, char** argv) {
    setvbuf(stdout, nullptr, _IONBF, 0); /* unbuffered: output must survive a timeout kill or AV */
    if (argc < 2) { printf("FAIL usage: harness <scenario>\n"); return 1; }

    HMODULE m = LoadLibraryA("BlnetTestShim.dll");
    if (!m) { printf("FAIL load: BlnetTestShim.dll (error %lu)\n", GetLastError()); return 2; }

    /* Contract exports - bound straight into the runtime's ShimApi so NetCheck /
       NetRef / blnet_get_native_error work. Names come from the BLNET_EXPORT_*
       macros in blnet.h, never string literals here. */
    bind(m, g_shim.abi_version, BLNET_EXPORT_ABI_VERSION);
    bind(m, g_shim.initialize,  BLNET_EXPORT_INITIALIZE);
    bind(m, g_shim.addref,      BLNET_EXPORT_ADDREF);
    bind(m, g_shim.release,     BLNET_EXPORT_RELEASE);
    bind(m, g_shim.alloc,       BLNET_EXPORT_ALLOC);
    bind(m, g_shim.free_,       BLNET_EXPORT_FREE);
    bind(m, g_shim.last_error,  BLNET_EXPORT_LAST_ERROR);
    shim_abi_version = g_shim.abi_version;
    shim_initialize  = g_shim.initialize;

    /* Test exports (shim-only surface, literal names by design). */
    bind(m, test_create_list,        "blnet_test_create_list");
    bind(m, test_list_add,           "blnet_test_list_add");
    bind(m, test_list_count,         "blnet_test_list_count");
    bind(m, test_echo,               "blnet_test_echo");
    bind(m, test_throw,              "blnet_test_throw");
    bind(m, test_invoke,             "blnet_test_invoke");
    bind(m, test_invoke_from_thread, "blnet_test_invoke_from_thread");

    const char* scenario = argv[1];

    /* Scenario 14 must own the FIRST blnet_initialize call in this process - its
       whole point is that a MISMATCHED version is rejected before any successful
       handshake ever happened - so it dispatches BEFORE the shared handshake block
       and performs its own mismatched-then-correct init sequence. Every other
       scenario takes the normal handshake path below. */
    /* NB: dispatches OUTSIDE main's try/catch - keep NetCheck/throwing calls out of scenario_version_mismatch */
    if (strcmp(scenario, "version_mismatch") == 0) return scenario_version_mismatch();

    /* C7 handshake: version agreement, then hand the shim the native vtable.
       Handshake failures are ENVIRONMENTAL (exit 2, like bind failures), not
       scenario failures (exit 1). */
    static BlnetNativeVtable vt{ &blnet_invoke_callback, &blnet_get_native_error };
    if (shim_abi_version() != BLNET_ABI_VERSION) { printf("FAIL init: shim ABI version != BLNET_ABI_VERSION\n"); return 2; }
    if (shim_initialize(BLNET_ABI_VERSION, &vt) != BLNET_OK) { printf("FAIL init: blnet_initialize failed\n"); return 2; }

    try {
        if (strcmp(scenario, "handle_roundtrip") == 0) return scenario_handle_roundtrip();
        if (strcmp(scenario, "stale_handle") == 0) return scenario_stale_handle();
        if (strcmp(scenario, "double_release_addref") == 0) return scenario_double_release_addref();
        if (strcmp(scenario, "generation_reuse") == 0) return scenario_generation_reuse();
        if (strcmp(scenario, "string_roundtrip") == 0) return scenario_string_roundtrip();
        if (strcmp(scenario, "managed_exception") == 0) return scenario_managed_exception();
        if (strcmp(scenario, "inline_native_exception") == 0) return scenario_inline_native_exception();
        if (strcmp(scenario, "inline_result_and_out") == 0) return scenario_inline_result_and_out();
        if (strcmp(scenario, "struct_slots") == 0) return scenario_struct_slots();
        if (strcmp(scenario, "queued_notification_addref") == 0) return scenario_queued_notification_addref();
        if (strcmp(scenario, "cross_thread_result_rejected") == 0) return scenario_cross_thread_result_rejected();
        if (strcmp(scenario, "pump_error_surfacing") == 0) return scenario_pump_error_surfacing();
        if (strcmp(scenario, "deadlock_guard") == 0) return scenario_deadlock_guard();
        /* version_mismatch (scenario 14) dispatches BEFORE the handshake above */
        if (strcmp(scenario, "concurrency_hammer") == 0) return scenario_concurrency_hammer();
        if (strcmp(scenario, "stale_callback_via_release") == 0) return scenario_stale_callback_via_release();
    } catch (const std::exception& ex) {
        printf("FAIL %s: unexpected exception: %s\n", scenario, ex.what());
        return 1;
    }
    printf("FAIL unknown scenario: %s\n", scenario);
    return 1;
}
